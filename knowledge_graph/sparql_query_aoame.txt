###############
# GIVEN A GUEST PROFILE, FIND ALL MEALS THEY CAN EAT BASED ON THEIR DIET TYPE, MAX CALORIES, AND ALLERGIES
###############

SELECT ?Meal (SUM(?cal) AS ?Calories) 
WHERE {
  mod:suggestMeal_4ff4dc2a-8de0-4d60-843f-50e64e55f8a7 
    #lo:dietType ?diet ;
    #lo:maxCalories ?maxCalories;
    #lo:allergy ?guestAllergy .
  	lo:guestName ?guest .
  
  
  #BIND( IRI(CONCAT(STR(:), ?diet)) AS ?dietIRI)
  #BIND( IRI(CONCAT(STR(:), ?maxCalories)) AS ?maxCaloriesIRI)
  #BIND( IRI(CONCAT(STR(:), ?guestAllergy)) AS ?guestAllergyIRI)
  BIND( IRI(CONCAT(STR(:), ?guest)) AS ?guestIRI ) .
    
  ?guestIRI :hasDietType ?diet ;
         :gMaxCalories ?maxCalories .

  OPTIONAL { ?guestIRI :hasAllergy ?guestAllergy . }
  
  ?diet :canEat ?dish .
  ?dish :name ?Meal ;
        :hasIngredient ?ingredient .
  
  ?ingredient :hasCalories ?cal .

  FILTER NOT EXISTS {
    ?guestIRI :hasAllergy ?guestAllergy .
    ?dish :hasIngredient ?dishIngredient .
    ?dishIngredient :containsAllergen ?guestAllergy .
  }
  
}
GROUP BY ?Meal ?maxCalories
HAVING (SUM(?cal) <= ?maxCalories)

###############
# GIVEN A GUEST DETAILS, FIND ALL MEALS THEY CAN EAT BASED ON THEIR DIET TYPE, MAX CALORIES, AND ALLERGIES
###############

SELECT ?Meal (SUM(?cal) AS ?Calories)
WHERE {
  mod:suggestMeal_cb141a48-eeb2-42b9-a047-275b79d4611c 
    lo:dietType ?diet ;
    lo:maxCalories ?maxCalories;
    lo:allergies ?guestAllergy .
  
  BIND( IRI(CONCAT(STR(:), ?diet)) AS ?dietIRI)
  
  BIND(
    IF(CONTAINS(LCASE(?guestAllergy), "gluten"),
       IRI(CONCAT(STR(:), "Gluten")),
       IRI(CONCAT(STR(:), ""))
    ) AS ?guestGlutenAllergyIRI
  )
  
  BIND(
    IF(CONTAINS(LCASE(?guestAllergy), "lactose"),
       IRI(CONCAT(STR(:), "Lactose")),
       IRI(CONCAT(STR(:), ""))
    ) AS ?guestLactoseAllergyIRI
  )
  
  BIND(
    IF(CONTAINS(LCASE(?guestAllergy), "fish"),
       IRI(CONCAT(STR(:), "Fish")),
       IRI(CONCAT(STR(:), ""))
    ) AS ?guestFishAllergyIRI
  )
  
  BIND(
    IF(CONTAINS(LCASE(?guestAllergy), "eggsallergen"),
       IRI(CONCAT(STR(:), "EggsAllergen")),
       IRI(CONCAT(STR(:), ""))
    ) AS ?guestEggsAllergyIRI
  )
  
  BIND(
    IF(CONTAINS(LCASE(?guestAllergy), "nuts"),
       IRI(CONCAT(STR(:), "Nuts")),
       IRI(CONCAT(STR(:), ""))
    ) AS ?guestNutsAllergyIRI
  )
  
  ?dietIRI :canEat ?dish .
  ?dish :name ?Meal ;
        :hasIngredient ?ingredient .

  ?ingredient :hasCalories ?cal .
  
  FILTER NOT EXISTS {
    ?dish :hasIngredient ?dishIngredient .
    ?dishIngredient :containsAllergen ?allergenFromDish .

    FILTER (
         (?allergenFromDish = ?guestGlutenAllergyIRI && ?guestGlutenAllergyIRI != IRI(CONCAT(STR(:), "")))
      || (?allergenFromDish = ?guestLactoseAllergyIRI && ?guestLactoseAllergyIRI != IRI(CONCAT(STR(:), "")))
      || (?allergenFromDish = ?guestFishAllergyIRI && ?guestFishAllergyIRI != IRI(CONCAT(STR(:), "")))
      || (?allergenFromDish = ?guestEggsAllergyIRI && ?guestEggsAllergyIRI != IRI(CONCAT(STR(:), "")))
      || (?allergenFromDish = ?guestNutsAllergyIRI && ?guestNutsAllergyIRI != IRI(CONCAT(STR(:), "")))
    )
  }
}
GROUP BY ?Meal ?maxCalories
HAVING (SUM(?cal) <= ?maxCalories)
